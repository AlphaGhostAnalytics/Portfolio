-- Rank customers within each contract type by how far their monthly
-- charge deviates from the segment average, from the Telco Churn project

WITH contract_stats AS (
    SELECT
        customer_id,
        contract_type,
        tenure_months,
        monthly_charges,
        churn_flag,
        AVG(monthly_charges) OVER (PARTITION BY contract_type) AS avg_segment_charge
    FROM telco_customers
),
ranked_customers AS (
    SELECT
        *,
        monthly_charges - avg_segment_charge AS charge_deviation,
        RANK() OVER (
            PARTITION BY contract_type
            ORDER BY monthly_charges DESC
        ) AS charge_rank
    FROM contract_stats
)
SELECT
    contract_type,
    customer_id,
    tenure_months,
    monthly_charges,
    ROUND(charge_deviation, 2) AS charge_deviation,
    charge_rank
FROM ranked_customers
WHERE charge_rank <= 10
ORDER BY contract_type, charge_rank;
import pandas as pd
import numpy as np

df = pd.read_csv('WA_Fn-UseC_-Telco-Customer-Churn.csv')

# Coerce blanks to NaN, then impute per contract-type median
df['TotalCharges'] = pd.to_numeric(df['TotalCharges'], errors='coerce')
df['TotalCharges'] = df.groupby('Contract')['TotalCharges'].transform(
    lambda x: x.fillna(x.median())
)

# Conditional feature engineering: flag high-risk tenure/contract combinations
df['high_risk_segment'] = np.where(
    (df['Contract'] == 'Month-to-month') & (df['tenure'] <= 6),
    1, 0
)

# Deviation of each customer's monthly charge from their contract-type average
df['charge_vs_segment_avg'] = (
    df['MonthlyCharges'] - df.groupby('Contract')['MonthlyCharges'].transform('mean')
)

df.to_csv('telco_features.csv', index=False)
print('Feature engineering complete.')
// Year-over-year change in churn rate, compared to the same period last year
Churn Rate YoY Change =
VAR CurrentChurnRate =
    DIVIDE(
        CALCULATE([Churned Customers], Customers[Churn] = "Yes"),
        CALCULATE([Total Customers])
    )
VAR PriorYearChurnRate =
    CALCULATE(
        [Churn Rate],
        SAMEPERIODLASTYEAR('Date'[Date])
    )
RETURN
    DIVIDE(CurrentChurnRate - PriorYearChurnRate, PriorYearChurnRate)
